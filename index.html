🎥Overview — video-utils.js10

Convert video buffers (MP4, WebM, GIF) to WebP stickers with duration trimming. Same filter chain as sticker-utils.js, but adds -ss (start time) and -t (duration) options.

💡 සිංහලෙන්: මේ file එක video එකක් (MP4, WebM) WebP sticker එකකට convert කරනවා. sticker-utils.js එකේ gifToSticker() එකට සමානයි, නමුත් අමතරව 2ක් තියෙනවා: (1) -ss 00:00:00 — video එකේ මුල ඉඳන් පටන් ගන්නවා, (2) -t 00:00:05 — first 5 seconds විතරයි ගන්නවා. const fs = require('fs');
const path = require('path');
const { tmpdir } = require('os');
const Crypto = require('crypto');
const ffmpegPath = require('@ffmpeg-installer/ffmpeg').path;
const ffmpeg = require('fluent-ffmpeg');

// Set ffmpeg path
ffmpeg.setFfmpegPath(ffmpegPath);

/**
 * Converts a video or GIF buffer to a WebP sticker format.
 * @param {Buffer} videoBuffer - The video or GIF buffer to convert.
 * @returns {Promise<Buffer>} - The converted WebP sticker buffer.
 */
async function videoToWebp(videoBuffer) {
  const outputPath = path.join(
    tmpdir(),
    Crypto.randomBytes(6).readUIntLE(0, 6).toString(36) + '.webp'
  );
  const inputPath = path.join(
    tmpdir(),
    Crypto.randomBytes(6).readUIntLE(0, 6).toString(36) + '.mp4'
  );

  // Save the video buffer to a file
  fs.writeFileSync(inputPath, videoBuffer);

  await new Promise((resolve, reject) => {
    ffmpeg(inputPath)
      .on('error', reject)
      .on('end', () => resolve(true))
      .addOutputOptions([
        '-vcodec', 'libwebp',
        '-vf', "scale='min(320,iw)':min'(320,ih)':force_original_aspect_ratio=decrease,fps=15,pad=320:320:-1:-1:color=white@0.0,split [a][b];[a] palettegen=reserve_transparent=on:transparency_color=ffffff [p];[b][p] paletteuse",
        '-loop', '0',
        '-ss', '00:00:00',
        '-t', '00:00:05',
        '-preset', 'default',
        '-an',
        '-vsync', '0'
      ])
      .toFormat('webp')
      .save(outputPath);
  });

  const webpBuffer = fs.readFileSync(outputPath);
  fs.unlinkSync(outputPath);
  fs.unlinkSync(inputPath);

  return webpBuffer;
}

module.exports = {
  videoToWebp
};

② Base36 filename — ⭐ අලුත්

Crypto.randomBytes(6).readUIntLE(0, 6).toString(36) + '.webp'
🔤 Base36 vs Base16 (hex)

sticker-utils.js — .toString('hex') → 12 chars (0-9, a-f)

video-utils.js — .readUIntLE(0,6).toString(36) → කෙටි, alphanumeric (0-9, a-z)

Base36 = 36 characters (10 digits + 26 letters) → කෙටි නම්, ලේසියෙන් කියවන්න පුළුවන්.

MethodFormatLengthSample
randomBytes(6).toString('hex')Base1612a3f9c2b8e1d4
randomBytes(6).readUIntLE(0,6).toString(36)Base36~8k9d2mx4q

readUIntLE(0, 6) — buffer එකේ offset 0 ඉඳන් 6 bytes කියවලා unsigned integer එකක් හදනවා (little-endian). ඊට පස්සේ .toString(36) — base36 string එකකට.

⚠️ 6 bytes = 48 bits

readUIntLE(0, 6) — JS Number එකට safe integer limit (53-bit) ඇතුළත. 6 bytes = 48 bits ✅ (7 bytes = 56 bits — 53-bit limit එක ඉක්මවනවා, precision loss). ඒ නිසා 6 bytes max ආරක්ෂිතයි.

2️⃣ WhatsApp Plugin — .tosticker (video reply)

const { videoToWebp } = require('../lib/video-utils');

module.exports = async (sock, m) => {
    try {
        if (!m.quoted) return m.reply('❗ Reply to a video');

        const type = m.quoted.mtype;
        if (!['videoMessage', 'imageMessage'].includes(type)) {
            return m.reply('❌ Reply to a video or image');
        }

        const mediaBuffer = await m.quoted.download();
        const stickerBuffer = await videoToWebp(mediaBuffer);

        await sock.sendMessage(m.from, {
            sticker: stickerBuffer
        }, { quoted: m });

    } catch (err) {
        console.error(err);
        m.reply('❌ Sticker creation failed');
    }
};

3️⃣ Trim to specific range (custom)

videoToWebp() එකේ -ss සහ -t hardcoded (0s → 5s). Custom range ඕන නම් function එකට parameters add කරන්න: 

async function videoToWebpRange(videoBuffer, startTime = '00:00:00', duration = '00:00:05') {
    const outputPath = path.join(tmpdir(), Crypto.randomBytes(6).toString('hex') + '.webp');
    const inputPath = path.join(tmpdir(), Crypto.randomBytes(6).toString('hex') + '.mp4');

    fs.writeFileSync(inputPath, videoBuffer);

    await new Promise((resolve, reject) => {
        ffmpeg(inputPath)
            .on('error', reject)
            .on('end', () => resolve(true))
            .addOutputOptions([
                '-vcodec', 'libwebp',
                '-vf', "scale='min(320,iw)':min'(320,ih)':force_original_aspect_ratio=decrease,fps=15,pad=320:320:-1:-1:color=white@0.0,split [a][b];[a] palettegen=reserve_transparent=on:transparency_color=ffffff [p];[b][p] paletteuse",
                '-loop', '0',
                '-ss', startTime,       // ← custom
                '-t', duration,          // ← custom
                '-preset', 'default',
                '-an',
                '-vsync', '0'
            ])
            .toFormat('webp')
            .save(outputPath);
    });

    const webpBuffer = fs.readFileSync(outputPath);
    fs.unlinkSync(outputPath);
    fs.unlinkSync(inputPath);
    return webpBuffer;
}

// Usage
const sticker = await videoToWebpRange(videoBuffer, '00:00:02', '00:00:03'); // 2s-5s
